# cad2ui 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把 DWG/DXF 画的 UI 草图转成 AI 可精确读取的布局树（`layout.json` + `preview.html` + `preview.png` + `report.md`）。

**Architecture:** 单向流水线 `读文件 → 抽实体 → 建树 → 推断类型 → 算布局约束 → 序列化/渲染`，每段一个模块、只认自己的输入类型。几何统一到画布坐标，同时保留归一化到 1440 宽的参考坐标供阈值判定；四份产物全部从同一棵 `Node` 树生成。

**Tech Stack:** Python 3.12、ezdxf 1.3+、Pillow 10+、pytest。

**Spec:** `docs/superpowers/specs/2026-11-11-cad2ui-design.md`

## Global Constraints

- 解释器：`E:\python\python_address\python.exe`（3.12），依赖只装这一个解释器。
- 依赖：`ezdxf>=1.3`、`Pillow>=10`；开发期 `pytest`。不引入 matplotlib / shapely / pandas。
- 代码只放在 `tools/cad2ui/`，**不修改** `src/`、`static/`、`desktop/` 任何文件。
- 所有 `rect` 数值 `round(v, 1)`；JSON 用 `json.dumps(..., sort_keys=True, ensure_ascii=False, indent=2)`。
- `id` 按树先序编号 `n1..nN`；同一输入两次运行 `layout.json` 必须逐字节一致。
- 判定阈值一律在「参考坐标」（图纸归一化到 1440 宽）下比较；`--width` 只影响输出画布。
- 文本文件 UTF-8 无 BOM；中文只出现在字符串/注释/文档里。
- 不做 GUI、不写 DWG、不拆图块、不递归目录。

## Review Focus

- **空图纸 / 只有文字没有框**：必须 `NoGeometry`（退出码 4）或输出纯文字树，不能除零崩溃。
- **中文块名、中文标签**：PNG 不能出豆腐块，HTML/JSON 不能乱码。
- **极端尺度**（图纸宽 0.001 或 10^7 单位）：缩放后不得出现除零、`inf`、负尺寸。
- **旋转/镜像 INSERT、非矩形闭合多段线**：包围盒与 `points` 必须落在正确位置，不能偏到原点。
- **填充块盖住文字**：fill 不能当父节点，按钮/文字不能被错误挂进背景块。

```python
from __future__ import annotations

from dataclasses import dataclass, field


@dataclass(frozen=True)
class Rect:
    x: float
    y: float
    w: float
    h: float

    @property
    def right(self) -> float:
        return self.x + self.w

    @property
    def bottom(self) -> float:
        return self.y + self.h

    @property
    def area(self) -> float:
        return max(0.0, self.w) * max(0.0, self.h)


@dataclass
class Style:
    fill: str | None = None
    stroke: str | None = None
    color: str | None = None
    font_size: float | None = None
    radius: float = 0.0
    rotation: float = 0.0


@dataclass

    kind: str                       # box | line | text | circle | fill | block

    entity_type: str
    layer: str
    rect: Rect                      # 画布坐标（px）

    text: str | None = None
    block: str | None = None

    is_rect: bool = True


@dataclass
class Warning:
```

```python
from __future__ import annotations

import argparse
from pathlib import Path

from core.errors import Cad2UiError, ConverterUnavailable, NoGeometry, ParseError


def build_parser() -> argparse.ArgumentParser:
    p = argparse.ArgumentParser(
        prog="cad2ui",
        description="把 CAD（DWG/DXF）界面草图转成 AI 可读的布局树与 HTML 预览",
    )
    p.add_argument("input", type=Path, help="输入文件（.dwg 或 .dxf）")
    p.add_argument("-o", "--out", type=Path, default=None, help="输出目录，默认 out/<文件名>")
    p.add_argument("--width", type=int, default=1440, help="画布宽度 px，默认 1440")
    p.add_argument("--map", dest="map_file", type=Path, default=None,
                   help="外部映射表 JSON（块名/图层名 → 组件类型）")
    p.add_argument("--layers", default=None, help="只读这些图层，逗号分隔，支持 UI-* 通配")
    p.add_argument("--ignore-layers", default=None, help="跳过这些图层，逗号分隔")
    p.add_argument("--title", default=None, help="页面标题，默认取文件名")
    p.add_argument("--no-png", action="store_true", help="跳过 PNG 渲染")
    p.add_argument("--keep-dxf", action="store_true", help="保留 DWG 转换出的中间 DXF")
    p.add_argument("--quiet", action="store_true", help="只输出错误")
    return p


def _split(value: str | None) -> list[str] | None:
    if not value:
        return None
    return [part.strip() for part in value.split(",") if part.strip()]


def parse_args(argv: list[str] | None = None) -> argparse.Namespace:
    ns = build_parser().parse_args(argv)
    ns.layers = _split(ns.layers)
    ns.ignore_layers = _split(ns.ignore_layers)
    return ns


def main(argv: list[str] | None = None) -> int:
    ns = parse_args(argv)
    try:
        raise NoGeometry(f"pipeline 尚未接入：{ns.input}")
    except Cad2UiError as exc:
        print(f"错误：{exc}")
        return exc.exit_code


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
from __future__ import annotations

import shutil
import subprocess
from dataclasses import dataclass
from fnmatch import fnmatchcase
from pathlib import Path

import ezdxf

from core.errors import ConverterUnavailable, ParseError

SKIP_TYPES = frozenset({"DIMENSION", "LEADER", "MULTILEADER", "ACAD_TABLE"})
SKIP_LAYERS = ("Defpoints",)

CONVERTER_HINT = (
    "未找到 DWG 转换器。任选其一：\n"
    "  1) 安装 ODA File Converter（免费）：https://www.opendesign.com/guestfiles/oda_file_converter\n"
    "  2) 在 CAD 里「另存为」DXF，然后直接传 .dxf 文件"
)


@dataclass
class LoadedDrawing:
    doc: ezdxf.document.Drawing
    via: str
    entity_count: int


def match_layer(name: str, patterns: list[str]) -> bool:
    # CAD 图层名大小写不敏感；统一转小写再匹配，保证 Windows/Linux 行为一致
    low = name.lower()
    return any(fnmatchcase(low, p.lower()) for p in patterns)


def _read_dxf(path: Path) -> ezdxf.document.Drawing:
    try:
        return ezdxf.readfile(str(path))
    except FileNotFoundError:
        raise ParseError(f"文件不存在：{path}") from None
    except ezdxf.DXFStructureError as exc:
        raise ParseError(f"DXF 结构损坏：{exc}") from exc


def _convert_dwg(src: Path, work_dir: Path, keep_dxf: bool):
    target = work_dir / f"{src.stem}.cad2ui.dxf"
    try:
        from ezdxf.addons import odafc
        if odafc.is_installed():
            doc = odafc.readfile(str(src))
            if keep_dxf:
                doc.saveas(str(target))
            return doc
    except ImportError:
        pass

    exe = shutil.which("dwg2dxf")
    if exe:
        target.parent.mkdir(parents=True, exist_ok=True)
        proc = subprocess.run([exe, "-y", "-o", str(target), str(src)],
                              capture_output=True, text=True)
        if proc.returncode == 0 and target.exists():
            doc = _read_dxf(target)
            if not keep_dxf:
                target.unlink(missing_ok=True)
            return doc

    raise ConverterUnavailable(CONVERTER_HINT)


def load_drawing(path: Path, work_dir: Path, keep_dxf: bool = False) -> LoadedDrawing:
    path = Path(path)
    suffix = path.suffix.lower()
    work_dir.mkdir(parents=True, exist_ok=True)

    if suffix == ".dxf":
        doc, via = _read_dxf(path), "dxf"
    elif suffix == ".dwg":
        doc, via = _convert_dwg(path, work_dir, keep_dxf), "dwg->dxf"
    else:
        raise ConverterUnavailable(
            f"不支持的输入类型 {suffix or '(无扩展名)'}；只接受 .dwg / .dxf。\n{CONVERTER_HINT}"
        )

    return LoadedDrawing(doc=doc, via=via, entity_count=len(doc.modelspace()))


def select_entities(doc, layers: list[str] | None, ignore_layers: list[str] | None):
    keep = []
    skip_layers = list(SKIP_LAYERS) + list(ignore_layers or [])
    for e in doc.modelspace():
        etype = e.dxftype()
        if etype in SKIP_TYPES:
            continue
        layer = e.dxf.get("layer", "0")
        if match_layer(layer, skip_layers):
            continue
        if layers and not match_layer(layer, layers):
            continue
        keep.append(e)
    return keep
```

```python
from __future__ import annotations

# ACI 1-9 → 与 CONVENTION.md 一致的语义色（不是原始 RGB，便于前端直接复用）
ACI_PALETTE: dict[int, str] = {
    1: "#F56C6C",   # 红：危险/删除
    2: "#E6A23C",   # 橙：警告
    3: "#67C23A",   # 绿：成功
    4: "#17BECF",   # 青：图表
    5: "#409EFF",   # 蓝：主色
    6: "#B37FEB",   # 紫：强调
    7: "#303133",   # 白/黑：正文
    8: "#909399",   # 灰：次要文字
    9: "#C0C4CC",   # 浅灰：边框/填充
}

FALLBACK = "#888888"


def resolve_aci(doc, entity) -> tuple[str | None, int]:
    """返回 (十六进制, ACI)；BYLAYER/BYBLOCK 已解析。0-9 之外返回 (None, aci)。"""
    aci = int(entity.dxf.get("color", 256))
    if aci == 256:
        layer = doc.layers.get(entity.dxf.get("layer", "0"))
        aci = int(layer.dxf.color)
    if aci == 0:
        aci = 7
    return ACI_PALETTE.get(aci), aci
```

```python
from __future__ import annotations

from ezdxf import bbox as ezbbox

from core.colors import FALLBACK, resolve_aci
from core.dxf_io import LoadedDrawing, select_entities
from core.errors import NoGeometry
from core.model import Canvas, Element, Rect, Style, Warning

KIND_BY_TYPE = {
    "INSERT": "block",
    "LWPOLYLINE": "box",
    "POLYLINE": "box",
    "LINE": "line",
    "CIRCLE": "circle",
    "ELLIPSE": "circle",
    "POINT": "circle",
    "TEXT": "text",
    "MTEXT": "text",
    "ATTRIB": "text",
    "SOLID": "fill",
    "HATCH": "fill",
}


def plain_text(entity) -> str:
    fn = getattr(entity, "plain_text", None)
    if callable(fn):
        try:
            return (fn() or "").strip()
        except Exception:
            pass
    return (entity.dxf.get("text", "") or "").strip()


def _entity_bbox(entity) -> tuple[float, float, float, float] | None:
    try:
        bb = ezbbox.extents([entity])
    except Exception:
        return None
    if not bb.has_data:
        return None
    return (bb.extmin.x, bb.extmin.y, bb.extmax.x, bb.extmax.y)


def _text_bbox(entity) -> tuple[float, float, float, float] | None:
    bb = _entity_bbox(entity)
    if bb is not None:
        return bb
    try:
        p = entity.dxf.insert
    except Exception:
        return None
    h = float(entity.dxf.get("height", entity.dxf.get("char_height", 2.5) or 2.5))
    width = max(h, 0.6 * h * max(1, len(plain_text(entity))))
    return (p.x, p.y, p.x + width, p.y + h)


def _polygon_points(entity) -> list[tuple[float, float]]:
    if entity.dxftype() == "LWPOLYLINE":
        return [(p[0], p[1]) for p in entity.get_points("xy")]
    return [(p.x, p.y) for p in entity.points()]


def _is_axis_rect(points: list[tuple[float, float]], tol: float) -> bool:
    if len(points) not in (4, 5):
        return False
    xs = sorted({round(p[0] / tol) for p in points})
    ys = sorted({round(p[1] / tol) for p in points})
    return len(xs) == 2 and len(ys) == 2


def extract(loaded: LoadedDrawing, width: int = 1440,
            layers: list[str] | None = None,
            ignore_layers: list[str] | None = None
            ) -> tuple[list[Element], list[Warning], Canvas]:
    entities = select_entities(loaded.doc, layers, ignore_layers)
    if not entities:
        raise NoGeometry("图纸里没有可用实体（检查图层过滤或图纸是否为空）")

    bounds = []
    for e in entities:
        bb = _text_bbox(e) if e.dxftype() in ("TEXT", "MTEXT", "ATTRIB") else _entity_bbox(e)
        if bb is not None:
            bounds.append(bb)
    if not bounds:
        raise NoGeometry("所有实体都没有可计算的包围盒")

    bx0 = min(b[0] for b in bounds)
    by0 = min(b[1] for b in bounds)
    bx1 = max(b[2] for b in bounds)
    by1 = max(b[3] for b in bounds)
    dw = max(bx1 - bx0, 1e-6)
    dh = max(by1 - by0, 1e-6)
    scale = width / dw
    ref_scale = 1440.0 / dw
    tol = max(dw, dh) * 1e-6

    def to_rect(x0, y0, x1, y1, s) -> Rect:
        return Rect(
            round((min(x0, x1) - bx0) * s, 1),
            round((by1 - max(y0, y1)) * s, 1),
            round(abs(x1 - x0) * s, 1),
            round(abs(y1 - y0) * s, 1),
        )

    elements: list[Element] = []
    warnings: list[Warning] = []

    for z, entity in enumerate(entities):
        etype = entity.dxftype()
        kind = KIND_BY_TYPE.get(etype)
        if kind is None:
            continue

        bb = _text_bbox(entity) if kind == "text" else _entity_bbox(entity)
        if bb is None:
            continue

        rect = to_rect(*bb, scale)
        ref_rect = to_rect(*bb, ref_scale)

        if rect.w < 1 and rect.h < 1:
            warnings.append(Warning("DEGENERATE_RECT", entity.dxf.handle, etype, rect,
                                    "缩放后宽高都小于 1px，已丢弃"))
            continue

        layer = entity.dxf.get("layer", "0")
        color, aci = resolve_aci(loaded.doc, entity)
        if color is None:
            color = FALLBACK
            warnings.append(Warning("COLOR_APPROX", entity.dxf.handle, etype, rect,
                                    f"ACI {aci} 超出 0-9 标准色，使用近似灰"))

        style = Style()
        text = None
        block = None
        points = None
        is_rect = True

        if kind == "text":
            text = plain_text(entity)
            height = float(entity.dxf.get("height", entity.dxf.get("char_height", 0)) or 0)
            style.color = color
            style.font_size = round(height * scale, 1) or None
            style.rotation = float(entity.dxf.get("rotation", 0) or 0)
        elif kind == "line":
            style.stroke = color
            pts = _polygon_points(entity) if etype in ("LWPOLYLINE", "POLYLINE") else None
            if pts is None:
                pts = [(entity.dxf.start.x, entity.dxf.start.y),
                       (entity.dxf.end.x, entity.dxf.end.y)]
            points = [(round((p[0] - bx0) * scale, 1), round((by1 - p[1]) * scale, 1))
                      for p in pts]
        elif kind == "fill":
            style.fill = color
        elif kind == "block":
            block = entity.dxf.get("name")
            texts = [a.dxf.text.strip() for a in entity.attribs if a.dxf.text.strip()]
            text = " / ".join(texts) or None
            style.stroke = color
            style.rotation = float(entity.dxf.get("rotation", 0) or 0)
        elif kind in ("box", "circle"):
            style.stroke = color
            if kind == "circle":
                style.radius = round(max(rect.w, rect.h) / 2, 1)
            if etype in ("LWPOLYLINE", "POLYLINE"):
                pts = _polygon_points(entity)
                closed = entity.closed if etype == "LWPOLYLINE" else entity.is_closed
                if not closed or len(pts) < 3:
                    kind = "line"
                    warnings.append(Warning("UNCLOSED_POLYLINE", entity.dxf.handle, etype,
                                            rect, "多段线未闭合，按线处理"))
                    points = [(round((p[0] - bx0) * scale, 1),
                               round((by1 - p[1]) * scale, 1)) for p in pts]
                elif not _is_axis_rect(pts, tol):
                    is_rect = False
                    points = [(round((p[0] - bx0) * scale, 1),
                               round((by1 - p[1]) * scale, 1)) for p in pts]
                    warnings.append(Warning("CUT_CORNER_BOX", entity.dxf.handle, etype,
                                            rect, "非 4 顶点矩形，已保留原始点列 points"))

        elements.append(Element(
            kind=kind, handle=entity.dxf.handle or "", entity_type=etype, layer=layer,
            rect=rect, ref_rect=ref_rect, text=text, block=block, points=points,
            style=style, z=z, is_rect=is_rect,
        ))

    if not elements:
        raise NoGeometry("图纸里没有可识别的界面实体")

    canvas = Canvas(width=width, height=round(dh * scale), scale=round(scale, 4))
    return elements, warnings, canvas
```

注意 `_is_axis_rect` 里用 `round(p[0] / tol)` 分组，tol 是绝对容差，
对 1e6 单位的图纸仍然按相对比例缩放，没问题；但 `round()` 对负坐标是对称的，可用。

```python
from __future__ import annotations

from core.model import Element, Node, Rect, Warning


def contains(outer: Rect, inner: Rect, tol: float = 1.0) -> bool:
    return (inner.x >= outer.x - tol and inner.y >= outer.y - tol
            and inner.right <= outer.right + tol
            and inner.bottom <= outer.bottom + tol)


def _overlap_area(a: Rect, b: Rect) -> float:
    w = min(a.right, b.right) - max(a.x, b.x)
    h = min(a.bottom, b.bottom) - max(a.y, b.y)
    return max(0.0, w) * max(0.0, h)


def build_tree(elements: list[Element], warnings: list[Warning]) -> list[Node]:
    nodes: dict[int, Node] = {}
    for e in elements:
        nodes[id(e)] = Node(
            kind=e.kind, layer=e.layer, block=e.block, rect=e.rect,
            style=e.style, points=e.points, z=e.z,
            text=e.text, handle=e.handle, entity_type=e.entity_type,
        )

    # 面积升序：第一个“包含该元素”的 box 就是最小的父框（spec 7.3.2）
    boxes = sorted((e for e in elements if e.kind == "box"),
                   key=lambda e: (e.ref_rect.area, e.z))

    # 1) 父节点只能是 box
    parent: dict[int, int] = {}
    for e in elements:
        for b in boxes:
            if b is e:
                continue
            if e.kind == "box" and b.ref_rect.area <= e.ref_rect.area:
                continue
            if contains(b.ref_rect, e.ref_rect):
                parent[id(e)] = id(b)
                break

    # 2) 部分重叠 → OVERLAP（包含关系不算）
    for i, a in enumerate(boxes):
        for b in boxes[i + 1:]:
            if _overlap_area(a.ref_rect, b.ref_rect) <= 0:
                continue
            if contains(a.ref_rect, b.ref_rect) or contains(b.ref_rect, a.ref_rect):
                continue
            warnings.append(Warning(
                "OVERLAP", a.handle, a.entity_type, a.rect,
                f"框 {a.handle} 与 {b.handle} 部分重叠，按面积大者当父节点",
            ))

    # 2.5) 没落进任何框的文字 → TEXT_OUTSIDE_BOX
    for e in elements:
        if e.kind != "text" or id(e) in parent:
            continue
        partial = any(_overlap_area(e.ref_rect, b.ref_rect) > 0 for b in boxes)
        suffix = "（与框部分重叠）" if partial else ""
        warnings.append(Warning(
            "TEXT_OUTSIDE_BOX", e.handle, e.entity_type, e.rect,
            f"文字未完全落在任何框内{suffix}",
        ))

    # 3) 挂树
    roots: list[Node] = []
    for e in elements:
        pid = parent.get(id(e))
        if pid is None:
            roots.append(nodes[id(e)])
        else:
            nodes[pid].children.append(nodes[id(e)])

    # 4) 文字归位：每个 box 取字号最大的文字当 label，其余留作子节点
    best: dict[int, tuple[tuple[float, float, float], Element]] = {}
    for e in elements:
        if e.kind != "text" or not e.text:
            continue
        pid = parent.get(id(e))
        if pid is None:
            continue
        key = (e.style.font_size or 0.0, -e.rect.y, -e.rect.x)
        if pid not in best or key > best[pid][0]:
            best[pid] = (key, e)

    for pid, (_, e) in best.items():
        node = nodes[pid]
        node.label = e.text
        node.hints.append(f"label:{e.handle}")
        consumed = nodes[id(e)]
        node.children = [c for c in node.children if c is not consumed]

    # 5) 同级按 z 排序
    for node in nodes.values():
        node.children.sort(key=lambda n: n.z)
    roots.sort(key=lambda n: n.z)

    return roots


def assign_ids(tree: list[Node]) -> list[Node]:
    flat: list[Node] = []
    counter = 0

    def walk(children: list[Node], parent_id: str | None, depth: int) -> None:
        nonlocal counter
        for node in sorted(children, key=lambda n: (n.z, n.rect.y, n.rect.x)):
            counter += 1
            node.id = f"n{counter}"
            node.parent_id = parent_id
            node.depth = depth
            flat.append(node)
            walk(node.children, node.id, depth + 1)

    walk(tree, None, 0)
    return flat
```

```python
def load_mapping(path: Path) -> dict:
    data = json.loads(Path(path).read_text(encoding="utf-8-sig"))
    if not isinstance(data, dict):
        raise ValueError("映射表必须是 JSON 对象")
    if "blocks" in data or "layers" in data:
        return {
            "blocks": dict(data.get("blocks") or {}),
            "layers": dict(data.get("layers") or {}),
        }
    return {"blocks": dict(data), "layers": {}}


def keyword_type(name: str) -> str | None:
    if not name:
        return None
    low = name.lower()
    for type_name, words in KEYWORDS:
        for word in words:
            if word.lower() in low:
                return type_name
    return None
```

```python
from __future__ import annotations

from core.model import Node, Warning
from core.style import LAYER_CONTRACT, keyword_type

CONF_LABEL_MAP = {"layer": 1.0, "map": 0.95, "keyword": 0.85, "size": 0.5, "unknown": 0.2}


def _lookup(table: dict, key: str | None) -> str | None:
    if not key:
        return None
    if key in table:
        return table[key]
    low = key.lower()
    for k, v in table.items():
        if k.lower() == low:
            return v
    return None


def _size_type(node: Node, k: float) -> str | None:
    r = node.rect
    if r.w >= 1260 * k:
        if r.y <= 80 * k:
            return "header"
        if r.bottom >= 820 * k:
            return "footer"
    if r.h <= 56 * k and 40 * k <= r.w <= 320 * k and node.label:
        return "button"
    if r.w >= 300 * k and r.h >= 120 * k:
        return "card"
    if r.w < 48 * k and r.h < 48 * k:
        return "icon"
    return None


def infer_type(node: Node, mapping: dict, canvas_width: int = 1440
               ) -> tuple[str, float, list[str]]:
    k = canvas_width / 1440.0
    hints: list[str] = []

    contract = LAYER_CONTRACT.get(node.layer)
    if contract:
        return contract, 1.0, [f"layer:{node.layer}"]

    blocks = (mapping or {}).get("blocks") or {}
    layers = (mapping or {}).get("layers") or {}

    hit = _lookup(blocks, node.block)
    if hit:
        return hit, 0.95, [f"map:block:{node.block}"]
    hit = _lookup(layers, node.layer)
    if hit:
        return hit, 0.95, [f"map:layer:{node.layer}"]

    if node.block:
        hints.append(f"block:{node.block}")
        hit = keyword_type(node.block)
        if hit:
            return hit, 0.85, hints + [f"keyword:block:{node.block}"]
    hit = keyword_type(node.layer)
    if hit:
        return hit, 0.85, hints + [f"keyword:layer:{node.layer}"]

    hit = _size_type(node, k)
    if hit:
        return hit, 0.5, hints + [f"size:{hit}"]

    return "unknown", 0.2, hints


def apply_types(tree: list[Node], mapping: dict, canvas_width: int,
                warnings: list[Warning]) -> None:
    stack = list(tree)
    while stack:
        node = stack.pop()
        node.type, node.confidence, node.hints = infer_type(node, mapping, canvas_width)

        if node.type == "unknown" and node.block:
            warnings.append(Warning(
                "UNKNOWN_BLOCK", node.handle, node.entity_type, node.rect,
                f"块 {node.block} 没有命中任何映射或关键词",
            ))
        if node.type in ("button", "input", "select") and not node.label:
            warnings.append(Warning(
                "EMPTY_LABEL", node.handle, node.entity_type, node.rect,
                f"{node.type} 内没有文字，无法确定用途",
            ))

        stack.extend(node.children)
```

```python
from __future__ import annotations

from statistics import median

from core.model import Node, Rect


def _intervals_overlap(a0, a1, b0, b1) -> bool:
    return min(a1, b1) - max(a0, b0) > 0


def _all_pairs_overlap(spans: list[tuple[float, float]]) -> bool:
    for i in range(len(spans)):
        for j in range(i + 1, len(spans)):
            if not _intervals_overlap(spans[i][0], spans[i][1], spans[j][0], spans[j][1]):
                return False
    return True


def _union(rects: list[Rect]) -> Rect:
    x0 = min(r.x for r in rects)
    y0 = min(r.y for r in rects)
    x1 = max(r.right for r in rects)
    y1 = max(r.bottom for r in rects)
    return Rect(x0, y0, x1 - x0, y1 - y0)


def _align_cross(parent: Rect, children: list[Rect], direction: str) -> str:
    if direction == "column":
        if all(abs(c.w - parent.w) <= 2 for c in children):
            return "stretch"
        left = [c.x - parent.x for c in children]
        right = [parent.right - c.right for c in children]
    else:
        if all(abs(c.h - parent.h) <= 2 for c in children):
            return "stretch"
        left = [c.y - parent.y for c in children]
        right = [parent.bottom - c.bottom for c in children]

    spread_start = max(left) - min(left)
    spread_end = max(right) - min(right)
    spread_center = max(abs(l - r) for l, r in zip(left, right))

    candidates = [(spread_start, 0, "start"), (spread_end, 1, "end"),
                  (spread_center, 2, "center")]
    return min(candidates, key=lambda c: (c[0], c[1]))[2]


def _justify_main(parent: Rect, children: list[Rect], direction: str,
                  gap: float | None) -> str | None:
    if direction is None:
        return None
    if direction == "row":
        start, end = parent.x, parent.right
        lead = min(c.x for c in children) - start
        trail = end - max(c.right for c in children)
    else:
        start, end = parent.y, parent.bottom
        lead = min(c.y for c in children) - start
        trail = end - max(c.bottom for c in children)

    lead, trail = max(0.0, lead), max(0.0, trail)
    if gap and gap > 0 and abs(lead - gap) <= 2 and abs(trail - gap) <= 2:
        return "space-between"
    if lead <= 2:
        return "start"
    if trail <= 2:
        return "end"
    if abs(lead - trail) <= 2:
        return "center"
    return "start"


def compute_container(node: Node) -> tuple[dict | None, float | None]:
    children = [c.rect for c in node.children]
    if not children:
        return None, None

    span = _union(children)
    parent = node.rect

    padding = [
        int(round(max(0.0, span.y - parent.y))),
        int(round(max(0.0, parent.right - span.right))),
        int(round(max(0.0, parent.bottom - span.bottom))),
        int(round(max(0.0, span.x - parent.x))),
    ]
    children_span = {"x": round(span.x, 1), "y": round(span.y, 1),
                     "w": round(span.w, 1), "h": round(span.h, 1)}

    if len(children) < 2:
        return {
            "direction": None, "justify": None, "align": None, "gap": None,
            "padding": padding, "childrenSpan": children_span,
        }, 0.3

    y_overlap = _all_pairs_overlap([(c.y, c.bottom) for c in children])
    x_overlap = _all_pairs_overlap([(c.x, c.right) for c in children])
    if y_overlap and not x_overlap:
        direction = "row"
    elif x_overlap and not y_overlap:
        direction = "column"
    else:
        direction = None

    gap = None
    conf = 0.9
    if direction:
        axis = 0 if direction == "row" else 1
        ordered = sorted(children, key=lambda c: (c.x if axis == 0 else c.y))
        gaps = []
        for prev, nxt in zip(ordered, ordered[1:]):
            if axis == 0:
                gaps.append(nxt.x - prev.right)
            else:
                gaps.append(nxt.y - prev.bottom)
        gap = max(0, int(round(median(gaps))))
        if max(gaps) - min(gaps) > 4:
            conf -= 0.2
        align = _align_cross(parent, children, direction)
    else:
        conf = 0.3
        align = "start"

    justify = _justify_main(parent, children, direction, gap)

    return {
        "direction": direction,
        "justify": justify,
        "align": align,
        "gap": gap,
        "padding": padding,
        "childrenSpan": children_span,
    }, round(conf, 2)


def apply_computed(tree: list[Node]) -> None:
    stack = list(tree)
    while stack:
        node = stack.pop()
        node.computed, node.computed_confidence = compute_container(node)
        stack.extend(node.children)
```

```python
from __future__ import annotations

import json
from pathlib import Path

from core.model import Layout, Node, Rect, Warning

SCHEMA = "cad2ui/layout@1"


def round1(value: float) -> float:
    return round(float(value), 1)


def _rect_dict(rect: Rect) -> dict:
    return {"x": round1(rect.x), "y": round1(rect.y),
            "w": round1(rect.w), "h": round1(rect.h)}


def _style_dict(node: Node) -> dict:
    s = node.style
    out = {}
    if s.fill:
        out["fill"] = s.fill
    if s.stroke:
        out["stroke"] = s.stroke
    if s.color:
        out["color"] = s.color
    if s.font_size:
        out["fontSize"] = round1(s.font_size)
    if s.radius:
        out["radius"] = round1(s.radius)
    if s.rotation:
        out["rotation"] = round1(s.rotation)
    return out


def _node_dict(node: Node) -> dict:
    out = {
        "id": node.id,
        "parentId": node.parent_id,
        "type": node.type,
        "kind": node.kind,
        "label": node.label,
        "layer": node.layer,
        "rect": _rect_dict(node.rect),
        "z": node.z,
        "depth": node.depth,
        "confidence": round(node.confidence, 2),
    }
    if node.block:
        out["block"] = node.block
    if node.text:
        out["text"] = node.text
    style = _style_dict(node)
    if style:
        out["style"] = style
    if node.points:
        out["points"] = [[round1(x), round1(y)] for x, y in node.points]
    if node.hints:
        out["hints"] = list(node.hints)
    if node.computed:
        out["computed"] = node.computed
        out["computedConfidence"] = node.computed_confidence
    if node.children:
        out["children"] = [_node_dict(c) for c in node.children]
    return out


def _warning_dict(w: Warning) -> dict:
    return {
        "code": w.code,
        "handle": w.handle,
        "entityType": w.entity_type,
        "rect": _rect_dict(w.rect) if w.rect else None,
        "message": w.message,
    }


def layout_to_dict(layout: Layout, source: dict) -> dict:
    return {
        "schema": SCHEMA,
        "title": layout.title,
        "source": {
            "file": source["file"],
            "via": source["via"],
            "entityCount": source["entityCount"],
        },
        "canvas": {
            "width": layout.canvas.width,
            "height": layout.canvas.height,
            "scale": layout.canvas.scale,
        },
        "tree": [_node_dict(n) for n in layout.tree],
        "nodes": [_node_dict(n) for n in layout.nodes],
        "warnings": [_warning_dict(w) for w in layout.warnings],
        "stats": dict(sorted(layout.stats.items())),
    }


def dump_json(data: dict, path: Path) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    text = json.dumps(data, sort_keys=True, ensure_ascii=False, indent=2)
    path.write_text(text + "\n", encoding="utf-8", newline="\n")
```

```python
from __future__ import annotations

import html as htmllib
from pathlib import Path

from core.model import Layout, Node

CENTERED_TYPES = {"button", "input", "select", "tab", "badge", "switch", "checkbox"}

CSS = """
* { box-sizing: border-box; }
body { margin: 0; padding: 24px; background: #eef1f6;
       font-family: "Microsoft YaHei", "PingFang SC", "Segoe UI", sans-serif; }
.meta { max-width: 1440px; margin: 0 auto 12px; color: #606266; font-size: 13px; }
.meta b { color: #303133; }
.canvas { position: relative; margin: 0 auto; background: #fff;
          outline: 1px solid #c8cdd6; overflow: hidden; }
.node { position: absolute; }
.label { white-space: pre-wrap; }
.label.center { position: absolute; inset: 0; display: flex;
                align-items: center; justify-content: center; }
.label.corner { position: absolute; left: 8px; top: 6px; }
.points { position: absolute; left: 0; top: 0; overflow: visible; }
"""


def _px(v: float) -> str:
    return f"{round(v, 1):g}px"


def _style_attr(node: Node) -> str:
    s = node.style
    parts = [f"left:{_px(node.rect.x)}", f"top:{_px(node.rect.y)}",
             f"width:{_px(max(node.rect.w, 1))}", f"height:{_px(max(node.rect.h, 1))}",
             f"z-index:{node.z}"]
    if s.fill:
        parts.append(f"background:{s.fill}")
    if s.stroke and node.kind in ("box", "block", "circle"):
        parts.append(f"border:1px solid {s.stroke}")
    if node.kind == "line":
        parts.append(f"background:{s.stroke or '#909399'}")
    if node.kind == "circle":
        parts.append("border-radius:50%")
    if s.rotation:
        parts.append(f"transform:rotate({s.rotation:g}deg)")
    return ";".join(parts)


def _points_svg(node: Node) -> str:
    w = max(node.rect.w, 1)
    h = max(node.rect.h, 1)
    local = " ".join(f"{round(x - node.rect.x, 1):g},{round(y - node.rect.y, 1):g}"
                     for x, y in node.points or [])
    color = node.style.stroke or node.style.fill or "#909399"
    if node.kind == "line":
        shape = f'<polyline points="{local}" fill="none" stroke="{color}" stroke-width="1"/>'
    else:
        fill = node.style.fill or "none"
        shape = (f'<polygon points="{local}" fill="{fill}" '
                 f'stroke="{node.style.stroke or color}" stroke-width="1"/>')
    return (f'<svg class="points" width="{_px(w)}" height="{_px(h)}" '
            f'viewBox="0 0 {round(w, 1):g} {round(h, 1):g}">{shape}</svg>')


def _label_html(node: Node) -> str:
    if not node.label or node.kind == "text":
        return ""
    size = node.style.font_size or 14
    color = node.style.color or "#303133"
    if node.type in CENTERED_TYPES:
        return (f'<div class="label center" style="font-size:{_px(size)};color:{color}">'
                f'{htmllib.escape(node.label)}</div>')
    return (f'<div class="label corner" style="font-size:{_px(size)};color:{color}">'
            f'{htmllib.escape(node.label)}</div>')


def _text_html(node: Node) -> str:
    size = node.style.font_size or 14
    color = node.style.color or "#303133"
    return (f'<div class="node" data-cad-node="{node.id}" data-type="{node.type}" '
            f'data-label-source="1" style="{_style_attr(node)}">'
            f'<span class="label" style="font-size:{_px(size)};color:{color}">'
            f'{htmllib.escape(node.text or node.label or "")}</span></div>')


def _node_html(node: Node) -> str:
    if node.kind == "text":
        return _text_html(node)
    inner = _points_svg(node) if node.points else ""
    inner += _label_html(node)
    return (f'<div class="node" data-cad-node="{node.id}" data-type="{node.type}" '
            f'data-parent="{node.parent_id or ''}" style="{_style_attr(node)}">'
            f'{inner}</div>')


def render_html(layout: Layout, source: dict, path: Path) -> None:
    title = htmllib.escape(layout.title)
    meta = (
        f"<b>{title}</b> · 来源 {htmllib.escape(str(source['file']))} "
        f"({htmllib.escape(str(source['via']))}) · 画布 "
        f"{layout.canvas.width}×{layout.canvas.height}px · 节点 {len(layout.nodes)} · "
        f"warnings {len(layout.warnings)}"
    )
    body = "".join(_node_html(n) for n in layout.nodes)
    doc = (
        "<!doctype html>\n<html lang=\"zh-CN\">\n<head>\n<meta charset=\"utf-8\">\n"
        f"<title>{title}</title>\n<style>{CSS}</style>\n</head>\n<body>\n"
        f'<div class="meta">{meta}</div>\n'
        f'<div class="canvas" style="width:{layout.canvas.width}px;'
        f'height:{layout.canvas.height}px">\n{body}\n</div>\n</body>\n</html>\n'
    )
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(doc, encoding="utf-8", newline="\n")
```

```python
from __future__ import annotations

from functools import lru_cache
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

from core.model import Layout, Warning

FONT_CANDIDATES = (
    r"C:\Windows\Fonts\msyh.ttc",
    r"C:\Windows\Fonts\msyhbd.ttc",
    r"C:\Windows\Fonts\simhei.ttf",
    "/System/Library/Fonts/PingFang.ttc",
    "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
)

CENTERED = {"button", "input", "select", "tab", "badge", "switch", "checkbox"}


@lru_cache(maxsize=64)
def _font(size: int):
    for path in FONT_CANDIDATES:
        try:
            if Path(path).exists():
                return ImageFont.truetype(path, size)
        except OSError:
            continue
    return ImageFont.load_default()


def load_font(size: int):
    return _font(max(1, int(size)))


def _rgba(hex_color: str | None, default: tuple[int, int, int, int]):
    if not hex_color:
        return default
    value = hex_color.lstrip("#")
    if len(value) != 6:
        return default
    try:
        return (int(value[0:2], 16), int(value[2:4], 16), int(value[4:6], 16), 255)
    except ValueError:
        return default


def render_png(layout: Layout, path: Path, warnings: list[Warning]) -> None:
    img = Image.new("RGBA", (max(1, layout.canvas.width), max(1, layout.canvas.height)),
                    (255, 255, 255, 255))
    draw = ImageDraw.Draw(img)

    for node in sorted(layout.nodes, key=lambda n: n.z):
        r = node.rect
        box = [r.x, r.y, r.right, r.bottom]
        if node.kind in ("box", "block", "circle", "fill"):
            fill = _rgba(node.style.fill, None) if node.style.fill else None
            outline = _rgba(node.style.stroke, None)
            if node.kind == "circle":
                draw.ellipse(box, fill=fill, outline=outline)
            elif fill or outline:
                draw.rectangle(box, fill=fill, outline=outline)
        elif node.kind == "line":
            color = _rgba(node.style.stroke, (144, 147, 153, 255))
            if node.points and len(node.points) >= 2:
                draw.line([tuple(p) for p in node.points], fill=color, width=1)
            else:
                draw.rectangle(box, fill=color)

        if node.label and node.kind != "text" and node.type in CENTERED:
            size = int(node.style.font_size or 14)
            draw.text((r.x + r.w / 2, r.y + r.h / 2), node.label,
                      fill=_rgba(node.style.color or node.style.stroke, (48, 49, 51, 255)),
                      font=load_font(size), anchor="mm")
        elif node.label and node.kind != "text":
            size = int(node.style.font_size or 14)
            draw.text((r.x + 8, r.y + 6), node.label,
                      fill=_rgba(node.style.color or node.style.stroke, (48, 49, 51, 255)),
                      font=load_font(size))

    for node in layout.nodes:
        if node.kind != "text" or not (node.text or node.label):
            continue
        size = int(node.style.font_size or 14)
        draw.text((node.rect.x, node.rect.y), node.text or node.label,
                  fill=_rgba(node.style.color, (48, 49, 51, 255)), font=load_font(size))

    path.parent.mkdir(parents=True, exist_ok=True)
    img.convert("RGB").save(path, format="PNG")
```

```python
from __future__ import annotations

from pathlib import Path

from core.model import Layout, Warning


def _computed_cell(node) -> str:
    c = node.computed
    if not c:
        return "-"
    direction = c.get("direction") or "-"
    return f"{direction} gap={c.get('gap')} pad={c.get('padding')}"


def _warning_lines(warnings: list[Warning]) -> list[str]:
    if not warnings:
        return ["无。"]
    lines = ["| code | handle | 位置 | 说明 |", "|---|---|---|---|"]
    for w in warnings:
        rect = f"({w.rect.x:g},{w.rect.y:g},{w.rect.w:g},{w.rect.h:g})" if w.rect else "-"
        lines.append(f"| {w.code} | {w.handle or '-'} | {rect} | {w.message} |")
    return lines


def render_report(layout: Layout, source: dict, path: Path) -> None:
    lines = [
        f"# {layout.title}",
        "",
        f"- 来源：`{source['file']}`（{source['via']}）",
        f"- 画布：{layout.canvas.width} × {layout.canvas.height} px"
        f"（scale {layout.canvas.scale}）",
        f"- 节点：{len(layout.nodes)}",
        f"- 实体：{source['entityCount']}",
        "",
        "## 统计",
        "",
    ]
    for key, value in sorted(layout.stats.items()):
        lines.append(f"- {key}: {value}")

    lines += ["", "## 元素清单", "",
              "| id | 类型 | 文字 | 位置 | 尺寸 | 置信度 | 布局约束 |",
              "|---|---|---|---|---|---|---|"]
    for node in layout.nodes:
        r = node.rect
        label = (node.label or node.text or "").replace("|", "\\|")
        size = f"{r.w:g}×{r.h:g}"
        lines.append(
            f"| {node.id} | {node.type} | {label} | ({r.x:g},{r.y:g}) | {size} "
            f"| {node.confidence:.2f} | {_computed_cell(node)} |"
        )

    lines += ["", "## 警告", ""] + _warning_lines(layout.warnings)
    lines += ["", "画图规范见 `CONVENTION.md`。", ""]

    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text("\n".join(lines), encoding="utf-8", newline="\n")
```

```python
from __future__ import annotations

import argparse
from pathlib import Path

from core.computed import apply_computed
from core.dxf_io import load_drawing
from core.errors import Cad2UiError, ConverterUnavailable, NoGeometry, ParseError
from core.extract import extract
from core.infer import apply_types
from core.layout import assign_ids, build_tree
from core.model import Layout
from core.render_html import render_html
from core.render_png import render_png
from core.report import render_report
from core.serialize import dump_json, layout_to_dict
from core.style import load_mapping


def build_parser() -> argparse.ArgumentParser:
    p = argparse.ArgumentParser(
        prog="cad2ui",
        description="把 CAD（DWG/DXF）界面草图转成 AI 可读的布局树与 HTML 预览",
    )
    p.add_argument("input", type=Path, help="输入文件（.dwg 或 .dxf）")
    p.add_argument("-o", "--out", type=Path, default=None, help="输出目录，默认 out/<文件名>")
    p.add_argument("--width", type=int, default=1440, help="画布宽度 px，默认 1440")
    p.add_argument("--map", dest="map_file", type=Path, default=None,
                   help="外部映射表 JSON（块名/图层名 → 组件类型）")
    p.add_argument("--layers", default=None, help="只读这些图层，逗号分隔，支持 UI-* 通配")
    p.add_argument("--ignore-layers", default=None, help="跳过这些图层，逗号分隔")
    p.add_argument("--title", default=None, help="页面标题，默认取文件名")
    p.add_argument("--no-png", action="store_true", help="跳过 PNG 渲染")
    p.add_argument("--keep-dxf", action="store_true", help="保留 DWG 转换出的中间 DXF")
    p.add_argument("--quiet", action="store_true", help="只输出错误")
    return p


def _split(value: str | None) -> list[str] | None:
    if not value:
        return None
    return [part.strip() for part in value.split(",") if part.strip()]


def parse_args(argv: list[str] | None = None) -> argparse.Namespace:
    ns = build_parser().parse_args(argv)
    ns.layers = _split(ns.layers)
    ns.ignore_layers = _split(ns.ignore_layers)
    return ns


def _stats(nodes) -> dict:
    counts: dict[str, int] = {}
    for node in nodes:
        counts[node.kind] = counts.get(node.kind, 0) + 1
    counts["total"] = len(nodes)
    return counts


def run(ns: argparse.Namespace) -> int:
    src = Path(ns.input)
    out_dir = Path(ns.out) if ns.out else Path("out") / src.stem
    out_dir.mkdir(parents=True, exist_ok=True)

    loaded = load_drawing(src, out_dir, keep_dxf=ns.keep_dxf)
    elements, warnings, canvas = extract(loaded, width=ns.width,
                                         layers=ns.layers,
                                         ignore_layers=ns.ignore_layers)
    tree = build_tree(elements, warnings)
    flat = assign_ids(tree)

    mapping = load_mapping(ns.map_file) if ns.map_file else {"blocks": {}, "layers": {}}
    apply_types(tree, mapping, canvas.width, warnings)
    apply_computed(tree)

    layout = Layout(
        title=ns.title or src.stem,
        canvas=canvas,
        tree=tree,
        nodes=flat,
        warnings=warnings,
        stats=_stats(flat),
    )
    source = {"file": src.name, "via": loaded.via, "entityCount": loaded.entity_count}

    dump_json(layout_to_dict(layout, source), out_dir / "layout.json")
    render_html(layout, source, out_dir / "preview.html")
    if not ns.no_png:
        render_png(layout, out_dir / "preview.png", warnings)
    render_report(layout, source, out_dir / "report.md")

    if not ns.quiet:
        print(f"完成：{out_dir}")
        print(f"  节点 {len(flat)} / 实体 {loaded.entity_count} / warnings {len(warnings)}")
    return 0


def main(argv: list[str] | None = None) -> int:
    ns = parse_args(argv)
    try:
        return run(ns)
    except Cad2UiError as exc:
        print(f"错误：{exc}")
        return exc.exit_code
    except Exception as exc:                     # 兜底，不让用户看裸 traceback
        print(f"未预期错误：{type(exc).__name__}: {exc}")
        return 1


if __name__ == "__main__":
    raise SystemExit(main())
```
